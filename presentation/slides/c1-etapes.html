# 5. le réseau des pods
kubectl create -f $BASE/tigera-operator.yaml && kubectl create -f $BASE/custom-resources.yaml
# 6. sur worker1 et worker2 (commande affichée par init)
sudo kubeadm join 10.10.0.10:6443 --token <token> --discovery-token-ca-cert-hash sha256:<hash>

Étapes détaillées dans guide-installation-cluster.md. Faire remarquer que kubeadm init crée les static pods dans /etc/kubernetes/manifests et les certificats dans /etc/kubernetes/pki : on s'en resservira aux chapitres 9 et 11.
